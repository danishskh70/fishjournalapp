import 'dart:convert';
import 'package:http/http.dart' as http;
import 'package:flutter_secure_storage/flutter_secure_storage.dart';

class ApiException implements Exception {
  final int code;
  final String message;
  ApiException(this.code, this.message);
  @override
  String toString() => message;
}

class Api {
  static const base = 'https://fish-journal.onrender.com';
  static const _s = FlutterSecureStorage();
  static String? _a, _r;
  static void Function()? onExpired;

  static Future<bool> hasSession() async {
    _a = await _s.read(key: 'a');
    _r = await _s.read(key: 'r');
    return _a != null;
  }

  static Future<void> _save(Map m) async {
    _a = m['access_token'];
    _r = m['refresh_token'] ?? _r;
    await _s.write(key: 'a', value: _a);
    await _s.write(key: 'r', value: _r);
  }

  static Future<void> login(String u, String p) async =>
      _save(await req('POST', '/api/v1/auth/login',
          body: {'username': u, 'password': p}, auth: false));
  static Future<void> logout() async {
    try {
      await req('POST', '/api/v1/auth/logout',
          body: _r == null ? null : {'refresh_token': _r});
    } catch (_) {}
    _a = null;
    _r = null;
    await _s.deleteAll();
  }

  static Future<bool> _refresh() async {
    if (_r == null) return false;
    try {
      await _save(await req('POST', '/api/v1/auth/refresh',
          body: {'refresh_token': _r}, auth: false));
      return true;
    } catch (_) {
      return false;
    }
  }

  static Future<dynamic> req(String method, String path,
      {Map<String, dynamic>? body,
      Map<String, String>? q,
      bool auth = true,
      bool retry = true}) async {
    final uri = Uri.parse(base + path)
        .replace(queryParameters: q == null || q.isEmpty ? null : q);
    final rq = http.Request(method, uri)
      ..headers['Content-Type'] = 'application/json';
    if (auth && _a != null) rq.headers['Authorization'] = 'Bearer $_a';
    if (body != null) rq.body = jsonEncode(body);
    http.Response r;
    try {
      r = await http.Response.fromStream(
          await rq.send().timeout(const Duration(seconds: 60)));
    } catch (_) {
      throw ApiException(
          0, 'Cannot reach the server. Check your internet and try again.');
    }
    if (r.statusCode == 401 && auth) {
      if (retry && await _refresh()) {
        return req(method, path, body: body, q: q, retry: false);
      }
      onExpired?.call();
      throw ApiException(401, 'Session expired. Please log in again.');
    }
    final data = r.body.isEmpty ? null : jsonDecode(utf8.decode(r.bodyBytes));
    if (r.statusCode >= 200 && r.statusCode < 300) return data;
    throw ApiException(r.statusCode, _msg(r.statusCode, data));
  }

  static String _msg(int c, dynamic d) {
    if (c >= 500) return 'Server problem. Please try again shortly.';
    if (d is Map) {
      final x = d['detail'] ?? d['message'];
      if (x is String) return x;
      if (x is List) {
        return x.map((e) => e is Map ? '${e['msg']}' : '$e').join('\n');
      }
    }
    return {
          403: 'Not permitted.',
          404: 'Record not found.',
          409: 'Already exists / conflict.',
          422: 'Invalid data.'
        }[c] ??
        'Error $c';
  }
}

dynamic pick(dynamic m, List<String> ks) {
  if (m is Map) {
    for (final k in ks) {
      final v = m[k];
      if (v != null && v is! Map && v is! List) return v;
    }
    for (final v in m.values) {
      final r = pick(v, ks);
      if (r != null) return r;
    }
  }
  return null;
}

List asList(dynamic d) => d is List
    ? d
    : (d is Map
        ? ((d['items'] ??
            d['data'] ??
            d['results'] ??
            d['entries'] ??
            d['transactions'] ??
            []) as List)
        : []);
String inr(dynamic v) =>
    '₹${NumberFormat.decimalPattern('en_IN').format(num.tryParse('${v ?? 0}') ?? 0)}';
String kg(dynamic v) => '${num.tryParse('${v ?? 0}') ?? 0} kg';
String niceDate(String? s) {
  final d = DateTime.tryParse(s ?? '');

String label(String k) {

class Fut extends StatefulWidget {
  final Future<dynamic> Function() load;
  final Widget Function(BuildContext, dynamic, VoidCallback) build;
  const Fut({super.key, required this.load, required this.build});
  @override
  State<Fut> createState() => _FutS();
}

class _FutS extends State<Fut> {
  late Future<dynamic> f = widget.load();
  void re() => setState(() => f = widget.load());
  @override
  Widget build(BuildContext c) => FutureBuilder(
      future: f,
      builder: (c, s) {
        if (s.connectionState != ConnectionState.done) {
          return const Center(child: CircularProgressIndicator());
        }
        if (s.hasError) {
          return Center(
              child: Padding(
                  padding: const EdgeInsets.all(24),
                  child: Column(mainAxisSize: MainAxisSize.min, children: [
                    const Icon(Icons.cloud_off,
                        size: 48, color: Colors.black38),
                    const SizedBox(height: 12),
                    Text('${s.error}', textAlign: TextAlign.center),
                    const SizedBox(height: 12),
                    FilledButton(onPressed: re, child: const Text('Retry'))
                  ])));
        }
        return widget.build(c, s.data, re);
      });
}

    }

class _TF extends State<TxForm> {
  List parties = [], fish = [];
  int? p, f;
  String mode = 'cash';
  String? saved;
  final q = TextEditingController(),
      r = TextEditingController(),
      paid = TextEditingController(text: '0');
  @override
  void initState() {
    super.initState();
    Future.wait([
      fetch(widget.sale ? '/api/v1/hotels' : '/api/v1/suppliers'),
      fetch('/api/v1/fish')
    ])
        .then((x) => setState(() {
              parties = x[0];
              fish = x[1];
            }))
        .catchError((e) {
      if (mounted) toast(context, '$e');
    });
  }

  Future<void> rate() async {
    if (!widget.sale || p == null || f == null) return;
    try {
      final l = await fetch('/api/v1/hotels/$p/rates');
      final m = l
          .cast<dynamic>()
          .firstWhere((e) => e['fish_id'] == f, orElse: () => null);
      saved = m == null ? null : '${m['rate']}';
      setState(() => r.text = saved ?? '');
    } catch (_) {}
  }

  Future<void> submit() async {
    if (p == null ||
        f == null ||
        d(q.text) <= Decimal.zero ||
        (!widget.sale || true) && r.text.trim().isEmpty && !widget.sale) {
      toast(context, 'Fill all fields');
      return;
    }
    final body = widget.sale
        ? {
            'hotel_id': p,
            'fish_id': f,
            'quantity': n(q.text),
            'journal_id': widget.jid,
            if (r.text.isNotEmpty && r.text != saved) 'rate': n(r.text)
          }
        : {
            'journal_id': widget.jid,
            'supplier_id': p,
            'fish_id': f,
            'quantity': n(q.text),
            'rate': n(r.text),
            'paid_now': n(paid.text),
            'payment_mode': mode
          };
    try {
      await Api.req(
          'POST', widget.sale ? '/api/v1/hotel-sales' : '/api/v1/purchases',
          body: body);
      if (mounted) Navigator.pop(context, true);
    } on ApiException catch (e) {
      if (mounted) toast(context, e.message);
    }
  }

  @override
  Widget build(BuildContext c) {
    final amt = d(q.text) * d(r.text);
    final due = amt - d(paid.text);
    return Column(mainAxisSize: MainAxisSize.min, children: [
      Text(widget.sale ? 'Hotel sale' : 'New purchase',
          style: const TextStyle(fontSize: 20, fontWeight: FontWeight.w700)),
      const SizedBox(height: 14),
      drop(widget.sale ? 'Hotel' : 'Supplier', p, parties, (v) {
        setState(() => p = v);
        rate();
      }),
      const SizedBox(height: 12),
      drop('Fish', f, fish, (v) {
        setState(() => f = v);
        rate();
      }),
      const SizedBox(height: 12),
      numField(q, 'Quantity (kg)', onChanged: (_) => setState(() {})),
      const SizedBox(height: 12),
      numField(r, 'Rate per kg (₹)', onChanged: (_) => setState(() {})),
      const SizedBox(height: 12),
      if (!widget.sale) ...[
        numField(paid, 'Paid now (₹)', onChanged: (_) => setState(() {})),
        const SizedBox(height: 12),
        SegmentedButton<String>(segments: const [
          ButtonSegment(
              value: 'cash', label: Text('Cash'), icon: Icon(Icons.payments)),
          ButtonSegment(
              value: 'upi', label: Text('UPI'), icon: Icon(Icons.qr_code))
        ], selected: {
          mode
        }, onSelectionChanged: (s) => setState(() => mode = s.first)),
        const SizedBox(height: 12)
      ],
      Panel(
          child: Column(children: [
        Row(mainAxisAlignment: MainAxisAlignment.spaceBetween, children: [

          Text('₹$amt',
              style: const TextStyle(fontWeight: FontWeight.w800, fontSize: 18))

class MasterPage extends StatefulWidget {
  final String title, path;
  const MasterPage({super.key, required this.title, required this.path});
  @override
  State<MasterPage> createState() => _MS();
}

class _MS extends State<MasterPage> {
  bool inactive = false;
  int v = 0;
  Future<void> add() async {
    final t = TextEditingController();
    final ok = await showDialog<bool>(
        context: context,
        builder: (_) => AlertDialog(
                title: Text('Add ${widget.title}'),
                content: TextField(
                    controller: t,
                    autofocus: true,
                    decoration: const InputDecoration(labelText: 'Name')),
                actions: [
                  TextButton(
                      onPressed: () => Navigator.pop(context, false),
                      child: const Text('Cancel')),
                  FilledButton(
                      style: FilledButton.styleFrom(
                          minimumSize: const Size(90, 44)),
                      onPressed: () => Navigator.pop(context, true),
                      child: const Text('Save'))
                ]));
    if (ok != true || t.text.trim().isEmpty) return;
    try {
      await Api.req('POST', widget.path, body: {'name': t.text.trim()});
      setState(() => v++);
    } on ApiException catch (e) {
      if (mounted) toast(context, e.message);
    }
  }

  @override
  Widget build(BuildContext c) => Scaffold(
      appBar: AppBar(title: Text(widget.title), actions: [
        Row(children: [
          const Text('Inactive'),
          Switch(
              value: inactive,
              onChanged: (x) => setState(() {
                    inactive = x;
                    v++;
                  }))
        ])
      ]),
      floatingActionButton: FloatingActionButton(
          backgroundColor: ocean,
          foregroundColor: Colors.white,
          onPressed: add,
          child: const Icon(Icons.add)),
      body: Fut(
          key: ValueKey(v),
          load: () =>
              fetch(widget.path, q: {if (inactive) 'include_inactive': 'true'}),
          build: (c, data, re) =>
              ListView(padding: const EdgeInsets.all(16), children: [
                for (final e in data as List)
                  Panel(
                      child: Row(children: [
                    Expanded(
                        child: Text('${e['name']}',
                            style: const TextStyle(
                                fontSize: 16, fontWeight: FontWeight.w600))),
                    Switch(
                        value: e['is_active'] != false,
                        onChanged: (x) async {
                          try {
                            await Api.req('PATCH', '${widget.path}/${e['id']}',
                                body: {'is_active': x});
                            setState(() => v++);
                          } on ApiException catch (er) {
                            if (c.mounted) toast(c, er.message);
                          }
                        })
                  ]))
              ])));
}

class PwSheet extends StatefulWidget {
  const PwSheet({super.key});
  @override
  State<PwSheet> createState() => _PW();
}

class _PW extends State<PwSheet> {
  final o = TextEditingController(), n_ = TextEditingController();
  Future<void> save() async {
    if (n_.text.length < 10) {
      toast(context, 'New password needs at least 10 characters');
      return;
    }
    try {
      await Api.req('POST', '/api/v1/auth/change-password',
          body: {'current_password': o.text, 'new_password': n_.text});
      if (mounted) {
        Navigator.pop(context);
        toast(context, 'Password changed');
      }
    } on ApiException catch (e) {
      if (mounted) toast(context, e.message);
    }
  }

  @override
  Widget build(BuildContext c) =>
      Column(mainAxisSize: MainAxisSize.min, children: [
        const Text('Change password',
            style: TextStyle(fontSize: 20, fontWeight: FontWeight.w700)),
        const SizedBox(height: 14),
        TextField(
            controller: o,
            obscureText: true,
            decoration: const InputDecoration(labelText: 'Current password')),
        const SizedBox(height: 12),
        TextField(
            controller: n_,
            obscureText: true,
            decoration: const InputDecoration(
                labelText: 'New password (10+ characters)')),
        const SizedBox(height: 16),
        FilledButton(onPressed: save, child: const Text('Update'))
      ]);
}
